14/// Build a metadata index for any supported source, dispatching on the path.
15///
16/// Bruker `.d` folders and MOBILion `.mbi` files carry the same three axes, so the
17/// rest of the viewer never needs to know which it is looking at.
18#[cfg(feature = "native")]
19pub fn load_meta_any(path: &str) -> anyhow::Result<meta::MetaIndex> {
20    #[cfg(feature = "mbi")]
21    if mbi::is_mbi_path(path) {
22        return mbi::load_meta(path);
23    }
24    meta::MetaIndex::load(path)
25}

27/// Pick the loader mode matching a run's source format.
28#[cfg(feature = "native")]
29pub fn loader_mode_for(meta: &meta::MetaIndex, frame_ids: Vec<u32>) -> loader::LoaderMode {
30    #[cfg(feature = "mbi")]
31    if mbi::is_mbi_path(&meta.data_path) {
32        // Carry the FrameInfo through: the MBI loader needs RT and the MS1/MS2 split,
33        // which live in per-frame HDF5 attributes the index has already read.
34        let wanted: std::collections::HashSet<u32> = frame_ids.iter().copied().collect();
35        let frames = meta
36            .frames
37            .iter()
38            .filter(|f| wanted.contains(&f.id))
39            .copied()
40            .collect();
41        return loader::LoaderMode::Mbi { path: meta.data_path.clone(), frames };
42    }
43    loader::LoaderMode::Real { path: meta.data_path.clone(), frame_ids }
44}

46/// Whether a source can carry DIA isolation windows at all.
47///
48/// MOBILion MAF acquisitions are quadrupole-free — there are no isolation windows
49/// to draw, and trying to read them would mean opening the path as a Bruker
50/// dataset, which it is not.
51#[cfg(feature = "native")]
52pub fn has_isolation_windows(path: &str) -> bool {
53    #[cfg(feature = "mbi")]
54    if mbi::is_mbi_path(path) {
55        return false;
56    }
57    let _ = path;
58    true
59}
